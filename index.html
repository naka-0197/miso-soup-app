Supabase テーブル作成SQL

Supabaseの「SQL Editor」に貼り付けてRunを実行してください：

create table if not exists miso_logs (
  id uuid default gen_random_uuid() primary key,
  created_at timestamptz default now(),
  title text not null,
  catchphrase text,
  mood text,
  condition text,
  season text,
  weather text,
  season_story text,
  ingredients jsonb,
  miso_type text,
  dashi_tip text,
  steps jsonb,
  cooking_wisdom text
);

alter table miso_logs enable row level security;
create policy "Allow all public operations" on miso_logs for all using (true) with check (true);
